% A run's level depends only on its record count. Fixed bounds keep both
% advances aligned.
\begin{tikzpicture}[x=1cm,y=1cm,font=\normalsize,
  run/.style={draw=ink,fill=ink!4,line width=.8pt,minimum height=.42cm,
    inner sep=0pt},
  new run/.style={run,fill=teal!15,draw=teal},
  tag/.style={font=\small,text=quiet}]
  \path[use as bounding box] (0,.3) rectangle (13.7,4.7);

  \foreach \level/\records/\y in {0/{$\le$15k}/3.8,1/{$\le$100k}/2.8,
                                 2/{$\le$1M}/1.8,3/{$\le$10M}/.8} {
    \node[anchor=west] at (0,\y) {L\level};
    \node[tag,anchor=west] at (.6,\y) {\records};
  }
  \foreach \x in {2.8,3.3,3.8,4.3}
    \node[run,minimum width=.35cm] at (\x,3.8) {};
  \foreach \x in {3.1,4.2}
    \node[run,minimum width=.9cm] at (\x,2.8) {};
  \node[run,minimum width=1.8cm] (l2a) at (3.5,1.8) {};
  \node[run,minimum width=1.8cm] (l2b) at (5.5,1.8) {};
  \node[run,minimum width=3.6cm] at (4.4,.8) {};

  % An inserted run goes straight to the level its size dictates.
  \node[new run,minimum width=1.8cm] (inserted) at (7.6,1.8) {};
  \draw[link,draw=teal] (7.6,4.5) -- (inserted.north);
  \node[tag,anchor=west,text=teal] at (7.7,4.3) {insert};

  % Merging three L2 runs yields a run that stays in L2.
  \only<2>{
    \node[draw=ink,dashed,line width=.8pt,rounded corners=2pt,inner sep=4pt,
      fit=(l2a)(l2b)(inserted)] (inputs) {};
    \node[run,minimum width=2.6cm] (output) at (11.9,1.8) {};
    \draw[link] (inputs.east) -- (output.west);
    \node[tag,anchor=north] at ($(inputs.east)!.5!(output.west)$) {merge};
  }
\end{tikzpicture}
