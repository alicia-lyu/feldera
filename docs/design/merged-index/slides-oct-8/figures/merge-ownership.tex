% Slide-8 runs, one row per level. Dots inside a run mark who references it.
\begin{tikzpicture}[x=1cm,y=1cm,font=\normalsize,
  run/.style={draw=ink,fill=ink!4,line width=.8pt,minimum height=.45cm,
    inner sep=0pt},
  faded/.style={draw=ink!45,fill=white}]
  \path[use as bounding box] (1.6,.3) rectangle (10.4,3.8);
  \newcommand{\refdot}[3]{\fill[#1] ($(#2.east)-(#3,0)$) circle (1.8pt)}

  % Referrers: before-state, after-state, and the head (newest version).
  \foreach \name/\hue/\y in {$C^-$/ink/3.3,$C^+$/teal/2.8,Head/plum/2.3} {
    \fill[\hue] (9.0,\y) circle (2.2pt);
    \node[anchor=west,text=\hue] at (9.15,\y) {\name};
  }

  \foreach \level/\y in {0/3.0,1/1.9,2/.8}
    \node[text=quiet] at (2.1,\y) {L\level};
  % After the merge, L0 runs leave the head but stay alive, drawn faded.
  \alt<2>{\tikzset{old run/.style={faded},new run/.style={faded,draw=teal!45}}}
    {\tikzset{old run/.style={},new run/.style={fill=teal!15,draw=teal}}}
  \node[run,minimum width=.7cm,old run] (old) at (3.0,3.0) {};
  \node[run,minimum width=.7cm,new run] (new) at (4.0,3.0) {};
  \node[run,minimum width=1.7cm] (mid1) at (3.4,1.9) {};
  \node[run,minimum width=1.7cm] (mid2) at (5.4,1.9) {};
  \node[run,minimum width=3.2cm] (large) at (4.4,.8) {};

  % Both snapshots keep their references across the merge.
  \foreach \run in {old,mid1,mid2,large} \refdot{ink}{\run}{.47};
  \foreach \run in {old,new,mid1,mid2,large} \refdot{teal}{\run}{.3};
  \foreach \run in {mid1,mid2,large} \refdot{plum}{\run}{.13};

  % Before the merge the head holds L0; afterward it holds m instead.
  \only<1>{\foreach \run in {old,new} \refdot{plum}{\run}{.13};}
  \only<2>{
    \node[run,minimum width=1.4cm] (merged) at (7.6,1.9) {$m$};
    \refdot{plum}{merged}{.13};
    \node[draw=ink,dashed,line width=.8pt,rounded corners=2pt,inner sep=4pt,
      fit=(old)(new)] (sources) {};
    \draw[link] (sources.east) -| (merged.north);
  }
\end{tikzpicture}
